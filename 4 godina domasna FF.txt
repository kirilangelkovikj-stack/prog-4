
#include <iostream>
#include <vector>
using namespace std;

int n;
vector<vector<int>> res;
vector<bool> visited;
vector<int> parent;
bool DFS(int u, int t) {
    visited[u] = true;

    if (u == t)
        return true;

    for (int v = 0; v < n; v++) {
        if (res[u][v] > 0 && !visited[v]) {
            parent[v] = u;

            if (DFS(v, t))
                return true;
        }
    }
    return false;
}

int FordFulkerson(int s, int t) {
    int maxFlow = 0;

    while (true) {
        visited.assign(n, false);
        parent.assign(n, -1);
        if (!DFS(s, t))
            break;
        int flow = 1000000000;
        for (int v = t; v != s; v = parent[v]) {
            int u = parent[v];

            if (res[u][v] < flow)
                flow = res[u][v];
        }

        for (int v = t; v != s; v = parent[v]) {
            int u = parent[v];

            res[u][v] -= flow;
            res[v][u] += flow;
        }
        maxFlow += flow;
    }
    return maxFlow;
}
int main() {
    int m;
    cin >> n >> m;
    res.assign(n, vector<int>(n, 0));
    for (int i = 0; i < m; i++) {
        int u, v, c;
        cin >> u >> v >> c;
        res[u][v] += c;
    }
    int s, t;
    cin >> s >> t;
    cout << FordFulkerson(s, t) << endl;
    return 0;
}
